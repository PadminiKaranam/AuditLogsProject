package com.persistent.audit.model;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.Table;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;

import lombok.Data;
import lombok.NoArgsConstructor;

@Data
@NoArgsConstructor
@Entity
@Table(name = "USERS")
public class User {

	@Id
	@GeneratedValue(strategy = GenerationType.IDENTITY)
	@Column(name = "user_id")
	private Long userId;

	@NotBlank
	@Size(max = 100)
	@Column(name = "username", nullable = false, unique = true, length = 100)
	private String username;

	@NotBlank
	@Size(max = 50)
	@Column(name = "user_type", nullable = false, length = 50)
	private String userType;

	@NotBlank
	@Size(max = 255)
	@Column(name = "user_email", nullable = false, unique = true, length = 255)
	private String userEmail;

	@Size(max = 255)
	@Column(name = "password", length = 255)
	private String password;

	@Size(max = 100)
	@Column(name = "tenant_id", length = 100)
	private String tenantId;

	public User(Long userId, String username, String userType, String userEmail, String password) {
		this(userId, username, userType, userEmail, password, null);
	}

	public User(Long userId, String username, String userType, String userEmail, String password, String tenantId) {
		this.userId = userId;
		this.username = username;
		this.userType = userType;
		this.userEmail = userEmail;
		this.password = password;
		this.tenantId = tenantId;
	}
}
